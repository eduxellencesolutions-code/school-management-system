   4: const QUEUE_KEY = 'offline_queue:attendance'
   5: 
   6: export interface QueuedAttendance {
   7:   id: string // local id, for removal after sync
   8:   organizationId: string
   9:   groupId: string
  10:   groupName: string
  11:   termId: string
  12:   sessionId: string
  13:   date: string
  14:   entries: { learnerId: string; status: 'present' | 'absent' | 'late' }[]
  15:   queuedAt: string
  16: }
  17: 
  18: async function readQueue(): Promise<QueuedAttendance[]> {
  19:   const raw = await AsyncStorage.getItem(QUEUE_KEY)
  20:   return raw ? JSON.parse(raw) : []
  21: }
  22: 
  23: async function writeQueue(queue: QueuedAttendance[]): Promise<void> {
  24:   await AsyncStorage.setItem(QUEUE_KEY, JSON.stringify(queue))
  25: }
  26: 
  27: export async function enqueueAttendance(item: Omit<QueuedAttendance, 'id' | 'queuedAt'>): Promise<void> {
  28:   const queue = await readQueue()
  29:   queue.push({ ...item, id: `${Date.now()}-${Math.random().toString(36).slice(2)}`, queuedAt: new Date().toISOString() })
  30:   await writeQueue(queue)
  31: }
  32: 
  33: export async function getQueuedAttendance(): Promise<QueuedAttendance[]> {
  34:   return readQueue()
  35: }
  36: 
  37: // Attempts to submit every queued item. Successes are removed; failures
  38: // (e.g. still offline, or a real server error) stay queued for the next attempt.
  39: // Returns counts so calling UI can show a result without needing to inspect the queue itself.
  40: export async function processAttendanceQueue(): Promise<{ synced: number; failed: number }> {
  41:   const queue = await readQueue()
  42:   if (queue.length === 0) return { synced: 0, failed: 0 }
  43: 
  44:   let synced = 0
  45:   const remaining: QueuedAttendance[] = []
  46: 
  47:   for (const item of queue) {
  48:     try {
  49:       await submitAttendance({
  50:         organizationId: item.organizationId, groupId: item.groupId, termId: item.termId,
  51:         sessionId: item.sessionId, date: item.date, entries: item.entries,
  52:       })
  53:       synced++
  54:     } catch {
  55:       remaining.push(item) // still failing (offline or real error) â€” keep for next attempt
  56:     }
  57:   }
  58: 
  59:   await writeQueue(remaining)
  60:   return { synced, failed: remaining.length }
  61: }

   5: // Permanent fix: this route used to build its own inline Supabase client,
   6: // which meant it was the ONE place in the codebase that didn't apply the
   7: // shared getCookieDomain() logic from lib/supabase/server.ts. That caused
   8: // every session created through this route (all parent magic-link logins,
   9: // password resets) to be written as a host-only cookie
  10: // (Domain=results.eduxellence.org) instead of the shared
  11: // Domain=.eduxellence.org cookie every other login path uses -- so a parent
  12: // session and a staff session with the same cookie name could silently
  13: // collide in the same browser.
  14: //
  15: // Fix: use the same createClient() everyone else uses, so there is exactly
  16: // ONE place in the whole app that ever decides cookie domain. This class of
  17: // bug can't recur without touching that single shared file.
  18: 
  19: export async function GET(request: NextRequest) {
  20:   const { searchParams } = new URL(request.url)
  21:   const code = searchParams.get('code')
  22:   const token_hash = searchParams.get('token_hash')
  23:   const type = searchParams.get('type')
  24: 
  25:   const supabase = await createClient()
  26: 
  27:   if (token_hash && type === 'recovery') {
  28:     const { error } = await supabase.auth.verifyOtp({ token_hash, type: 'recovery' })
  29:     if (!error) {
  30:       return NextResponse.redirect(new URL('/auth/reset-password', request.url))
  31:     }
  32:     return NextResponse.redirect(new URL('/auth/reset-password?error=invalid_token', request.url))
  33:   }
  34: 
  35:   if (token_hash && type === 'magiclink') {
  36:     const { data: sessionData, error } = await supabase.auth.verifyOtp({ token_hash, type: 'magiclink' })
  37:     if (!error) {
  38:       const isParentAccount = sessionData?.user?.user_metadata?.is_parent_account === true
  39:       const destination = isParentAccount ? '/parent/dashboard' : '/dashboard'
  40:       return NextResponse.redirect(new URL(destination, request.url))
  41:     }
  42:     return NextResponse.redirect(new URL('/access?error=invalid_code', request.url))
  43:   }
  44: 
  45:   if (code) {
  46:     const { data: sessionData, error } = await supabase.auth.exchangeCodeForSession(code)
  47:     if (!error) {
  48:       const isParentAccount = sessionData?.user?.user_metadata?.is_parent_account === true
  49:       const destination = isParentAccount ? '/parent/dashboard' : '/dashboard'
  50:       return NextResponse.redirect(new URL(destination, request.url))
  51:     }
  52:   }
  53: 
  54:   return NextResponse.redirect(new URL('/login?error=auth_failed', request.url))
  55: }

  85: export async function POST(request: Request) {
  86:   const secret = request.headers.get('x-signup-notification-secret');
  87:   if (!secret || secret !== process.env.SIGNUP_NOTIFICATION_WEBHOOK_SECRET) {
  88:     return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  89:   }
  90: 
  91:   let outboxId: string | undefined;
  92:   try {
  93:     ({ outboxId } = await request.json());
  94:   } catch {
  95:     return NextResponse.json({ error: 'Invalid JSON body' }, { status: 400 });
  96:   }
  97:   if (!outboxId) return NextResponse.json({ error: 'Missing outboxId' }, { status: 400 });
  98: 
  99:   const admin = createAdminClient();
 100: 
 101:   const { data: row, error: fetchError } = await admin
 102:     .from('signup_notification_outbox')
 103:     .select('*')
 104:     .eq('id', outboxId)
 105:     .maybeSingle();
 106: 
 107:   if (fetchError || !row) {
 108:     return NextResponse.json({ error: 'Outbox row not found' }, { status: 404 });
 109:   }
 110: 
 111:   // Idempotent guards â€” never re-send a completed or permanently
 112:   // exhausted notification, no matter how this route got invoked.
 113:   if (row.status === 'sent' || row.status === 'exhausted') {
 114:     return NextResponse.json({ success: true, skipped: row.status });
 115:   }
 116: 
 117:   const { subject, html } = buildEmail(row.signup_type, row.payload);
 118:   const attemptCount = (row.attempt_count ?? 0) + 1;
 119: 
 120:   try {
 121:     if (!process.env.RESEND_API_KEY) {
 122:       throw new Error('RESEND_API_KEY not configured');
 123:     }
 124:     const { Resend } = await import('resend');
 125:     const resend = new Resend(process.env.RESEND_API_KEY);
 126: 
 127:     await resend.emails.send({
 128:       from: 'Eduxellence Results <notifications@eduxellence.org>',
 129:       to: process.env.ADMIN_NOTIFICATION_EMAIL || 'j.sylvester@eduxellence.org',
 130:       subject,
 131:       html,
 132:     });
 133: 
 134:     await admin
 135:       .from('signup_notification_outbox')
 136:       .update({
 137:         status: 'sent',
 138:         sent_at: new Date().toISOString(),
 139:         attempt_count: attemptCount,
 140:         last_attempt_at: new Date().toISOString(),
 141:       })
 142:       .eq('id', outboxId);
 143: 
 144:     return NextResponse.json({ success: true });
 145:   } catch (err) {
 146:     console.error('signup-notification dispatch failed:', err);
 147:     const errorMessage = err instanceof Error ? err.message : String(err);
 148:     const backoff = nextBackoff(attemptCount);
 149: 
 150:     await admin
 151:       .from('signup_notification_outbox')
 152:       .update({
 153:         status: backoff ? 'failed' : 'exhausted',
 154:         attempt_count: attemptCount,
 155:         last_attempt_at: new Date().toISOString(),
 156:         next_attempt_at: backoff,
 157:         last_error: errorMessage,
 158:       })
 159:       .eq('id', outboxId);
 160: 
 161:     // 200, not 5xx: the failure is durably recorded for retry/investigation;
 162:     // we don't want the caller (pg_net or the cron retry loop) treating
 163:     // this as a transport error and doing its own separate retries.
 164:     return NextResponse.json({ success: false, error: 'Email send failed, logged for retry' }, { status: 200 });
 165:   }
 166: }
