   1: import { createClient } from '@/lib/supabase/server';

   5: // POST /api/attendance
   6: // Body: { groupId, termId, sessionId, date, records: [{ learnerId, status }] }
   7: export async function POST(request: Request) {
   8:   const supabase = await createClient();
   9: 
  10:   const { user } = await getAuthenticatedUser(supabase);
  11:   if (!user) {
  12:     return NextResponse.json({ error: 'Not authenticated' }, { status: 401 });
  13:   }
  14: 
  15:   // âœ… FIX: Check permission instead of hard role check
  16:   const { data: userRow } = await supabase
  17:     .from('users')
  18:     .select('organization_id, role')
  19:     .eq('id', user.id)
  20:     .single();
  21: 
  22:   const { data: hasPerm } = await supabase.rpc('has_permission', { 
  23:     p_user_id: user.id, 
  24:     p_permission_key: 'attendance.mark' 
  25:   });
  26:   if (userRow?.role !== 'admin' && !hasPerm) {
  27:     return NextResponse.json({ error: 'You do not have permission to mark attendance' }, { status: 403 });
  28:   }
  29: 
  30:   const body = await request.json();
  31:   const { groupId, termId, sessionId, date, records } = body;
  32: 
  33:   if (!groupId || !termId || !sessionId || !date || !Array.isArray(records) || records.length === 0) {
  34:     return NextResponse.json({ error: 'Missing required fields' }, { status: 400 });
  35:   }
  36: 
  37:   // Resolve the caller's org (null org = solo teacher, always allowed)
  38:   const { data: userRowOrg, error: userError } = await supabase
  39:     .from('users')
  40:     .select('organization_id')
  41:     .eq('id', user.id)
  42:     .single();
  43: 
  44:   if (userError) {
  45:     return NextResponse.json({ error: 'Could not resolve user profile' }, { status: 500 });
  46:   }
  47: 
  48:   const isSolo = userRowOrg.organization_id === null;
  49: 
  50:   // Explicit app-layer gate check, in addition to RLS â€” belt and braces.
  51:   if (!isSolo) {
  52:     const { data: hasFeature, error: featureError } = await supabase
  53:       .rpc('org_has_feature', {
  54:         p_org_id: userRowOrg.organization_id,
  55:         p_feature_key: 'basic_attendance',
  56:       });
  57: 
  58:     if (featureError) {
  59:       return NextResponse.json({ error: 'Could not verify plan entitlement' }, { status: 500 });
  60:     }
  61: 
  62:     if (!hasFeature) {
  63:       return NextResponse.json(
  64:         { error: 'Attendance is not available on your current plan' },
  65:         { status: 403 }
  66:       );
  67:     }
  68:   }
  69: 
  70:   // A withdrawn/transferred/graduated/suspended/archived student must
  71:   // not accumulate new attendance records â€” this is the exact contamination
  72:   // risk called out for attendance specifically. Filter the incoming
  73:   // learner IDs down to currently active ones before writing anything.
  74:   const learnerIds = [...new Set(records.map((r: { learnerId: string }) => r.learnerId))];
  75:   const { data: activeLearners, error: activeLearnersError } = await supabase
  76:     .from('learners')
  77:     .select('id')
  78:     .in('id', learnerIds)
  79:     .eq('is_active', true);
  80:   if (activeLearnersError) {
  81:     return NextResponse.json({ error: activeLearnersError.message }, { status: 500 });
  82:   }
  83:   const activeLearnerIds = new Set((activeLearners ?? []).map((l) => l.id));
  84:   const skippedLearnerIds = learnerIds.filter((id) => !activeLearnerIds.has(id));
  85: 
  86:   // organization_id is stamped from resolved user context, never trusted from client input
  87:   const rows = records
  88:     .filter((r: { learnerId: string; status: string }) => activeLearnerIds.has(r.learnerId))
  89:     .map((r: { learnerId: string; status: string }) => ({
  90:       organization_id: userRowOrg.organization_id,
  91:       group_id: groupId,
  92:       learner_id: r.learnerId,
  93:       term_id: termId,
  94:       session_id: sessionId,
  95:       date,
  96:       status: r.status,
  97:       recorded_by: user.id,
  98:     }));
  99: 
 100:   if (rows.length === 0) {
 101:     return NextResponse.json({ error: 'No active students among the submitted records' }, { status: 400 });
 102:   }
 103: 
 104:   const { error: insertError } = await supabase
 105:     .from('attendance_records')
 106:     .upsert(rows, { onConflict: 'learner_id,date' });
 107:   if (insertError) {
 108:     return NextResponse.json({ error: insertError.message }, { status: 500 });
 109:   }
 110: 
 111:   return NextResponse.json({ success: true, count: rows.length, skipped: skippedLearnerIds });
 112: }

  18: const SHARED_SECRET = process.env.RESULTS_CENTRAL_SHARED_SECRET!

  20: function verifySignature(rawBody: string, signatureHeader: string | null): boolean {
  21:   if (!signatureHeader || !SHARED_SECRET) return false
  22:   const computed = crypto.createHmac('sha256', SHARED_SECRET).update(rawBody).digest('hex')
  23:   // timing-safe compare
  24:   const a = Buffer.from(computed)
  25:   const b = Buffer.from(signatureHeader)
  26:   if (a.length !== b.length) return false
  27:   return crypto.timingSafeEqual(a, b)
  28: }

  17:   currency: string

  19:   const admin = serviceClient()
  20: 
  21:   // Re-verify amount/currency server-side against the live campaign price â€”
  22:   // never trust metadata.expected_amount alone, in case the qualifying

  35:   }

  37:   // Referral code is deliberately NOT taken from the frontend/metadata â€”
  38:   // enroll_founding_500() itself looks up the matching referrals row by
  39:   // organization_id server-side. We only need to know which one to pass.
  40:   const { data: referral } = await admin
  41:     .from('referrals')
  42:     .select('referral_code')
  43:     .eq('organization_id', metadata.organizationId)
  44:     .eq('status', 'pending')
  45:     .maybeSingle()
  46: 
  47:   if (!referral) {
  48:     console.error(`No pending referral found for organization ${metadata.organizationId} â€” cannot enroll in Founding 500`)
  49:     return { success: false, error: 'No valid referral found for this school. Please contact support.' }
  50:   }
  51: 
  52:   // âœ… enroll_founding_500() handles its own idempotency â€” no pre-claim needed
  53:   const { error: enrollError } = await admin.rpc('enroll_founding_500', {
  54:     p_org_id: metadata.organizationId,
  55:     p_referral_code: referral.referral_code,
  56:     p_payment_reference: reference,
  57:     p_provider: provider,
  58:     p_amount_paid: amountPaid,
  59:   })
  60: 
  61:   if (enrollError) {
  62:     console.error('enroll_founding_500 failed:', enrollError.message)
  63:     return { success: false, error: enrollError.message }
  64:   }
  65: 
  66:   return { success: true }
  67: }

  44:       return { success: true }

  47:     return { success: false, error: claimError.message }
  48:   }
  49:

  56: 
  57:   const updates = {
  58:     subscription_plan: metadata.plan,
  59:     subscription_status: 'active',

  61:     subscription_expires_at: expiresAt.toISOString(),

  86:

  90:           p_referral_id: referral.id,
  91:           p_plan: metadata.plan,
  92:           p_amount: amountPaid,
  93:         })
  94:       }
  95:     }
  96:   } catch (err) {
  97:     console.error('Commission recording failed (non-fatal):', err)
  98:   }
  99: 
 100:   return { success: true }
 101: }
