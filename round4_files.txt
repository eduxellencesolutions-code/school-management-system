  27: export async function getUsageCounts(ref: AccountRef): Promise<UsageCounts> {
  28:   const admin = serviceClient()
  29: 
  30:   if (ref.type === 'org') {
  31:     const [
  32:       { count: students },
  33:       { count: teachers },
  34:       { count: admins },
  35:       { count: academicSessions },
  36:       { count: classes },
  37:       { count: subjects },
  38:     ] = await Promise.all([
  39:       admin.from('learners').select('id', { count: 'exact', head: true }).eq('organization_id', ref.orgId),
  40:       admin.from('users').select('id', { count: 'exact', head: true }).eq('organization_id', ref.orgId).eq('role', 'teacher'),
  41:       admin.from('users').select('id', { count: 'exact', head: true }).eq('organization_id', ref.orgId).eq('role', 'admin'),
  42:       admin.from('academic_sessions').select('id', { count: 'exact', head: true }).eq('organization_id', ref.orgId),
  43:       admin.from('groups').select('id', { count: 'exact', head: true }).eq('organization_id', ref.orgId),
  44:       admin.from('subjects').select('id', { count: 'exact', head: true }).eq('organization_id', ref.orgId),
  45:     ])
  46: 
  47:     return {
  48:       students: students ?? 0,
  49:       teachers: teachers ?? 0,
  50:       admins: admins ?? 0,
  51:       academicSessions: academicSessions ?? 0,
  52:       classes: classes ?? 0,
  53:       subjects: subjects ?? 0,
  54:     }
  55:   }
  56: 
  57:   // Solo teacher: subjects/academic_sessions/groups are scoped directly
  58:   // by instructor_id. learners has no instructor_id column, so it must
  59:   // be counted via a join through groups (a student belongs to a group,
  60:   // and a group belongs to the teacher).
  61:   const [
  62:     { count: academicSessions },
  63:     { data: teacherGroups },
  64:     { count: subjects },
  65:   ] = await Promise.all([
  66:     admin.from('academic_sessions').select('id', { count: 'exact', head: true }).eq('instructor_id', ref.userId),
  67:     admin.from('groups').select('id').eq('instructor_id', ref.userId),
  68:     admin.from('subjects').select('id', { count: 'exact', head: true }).eq('instructor_id', ref.userId),
  69:   ])
  70: 
  71:   const groupIds = (teacherGroups ?? []).map(g => g.id)
  72: 
  73:   const { count: students } = groupIds.length > 0
  74:     ? await admin.from('learners').select('id', { count: 'exact', head: true }).in('group_id', groupIds)
  75:     : { count: 0 }
  76: 
  77:   return {
  78:     students: students ?? 0,
  79:     teachers: 1, // Solo teacher counts as their own teacher
  80:     admins: 1, // Solo teacher counts as their own admin
  81:     academicSessions: academicSessions ?? 0,
  82:     classes: groupIds.length,
  83:     subjects: subjects ?? 0,
  84:   }
  85: }

 185:

 188:   if (!allowed) return { success: false, message: message!, imported: 0, failed: 0 }
 189: 
 190:   // âœ… PLAN LIMIT GATE: Check maxStudents limit before importing
 191:   const limitCheck = await checkPlanLimit(supabase, user.id, 'maxStudents')
 192:   if (!limitCheck.allowed) {
 193:     return { success: false, message: limitCheck.message!, imported: 0, failed: 0 }
 194:   }
 195:

 197:   const rowsJson = formData.get('rows') as string
 198:   if (!groupId || !rowsJson) return { success: false, message: 'Missing data', imported: 0, failed: 0 }

 202:     rows = JSON.parse(rowsJson)

 204:     return { success: false, message: 'Invalid data format', imported: 0, failed: 0 }
 205:   }
