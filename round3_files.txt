######## src\lib\auth\getSchoolPermissions.ts ########
   1: import { SupabaseClient } from '@supabase/supabase-js'
   2: 
   3: /**
   4:  * Resolves a school-level user's full permission set, mirroring the
   5:  * has_permission() SQL function's exact logic (admin = universal bypass,
   6:  * everyone else = union of permissions granted via their active role
   7:  * assignments). Used to drive sidebar visibility -- NOT a substitute for
   8:  * the per-request has_permission() RPC check that every API route and page
   9:  * must still perform. The sidebar is a navigation aid; the real security
  10:  * boundary stays server-side on each route/page.
  11:  */
  12: export async function getSchoolPermissions(
  13:   supabase: SupabaseClient,
  14:   userId: string,
  15:   role: string | null | undefined
  16: ): Promise<{ isAdmin: boolean; permissions: string[] }> {
  17:   if (role === 'admin') {
  18:     return { isAdmin: true, permissions: [] }
  19:   }
  20: 
  21:   // Two-step query â€” PostgREST cannot resolve a direct relationship between
  22:   // staff_role_assignments and role_permissions (the FK path goes through
  23:   // school_roles), so the previous embedded-resource query always returned
  24:   // an error and silently yielded an empty permission set.
  25:   const { data: assignments, error: assignmentsError } = await supabase
  26:     .from('staff_role_assignments')
  27:     .select('role_id')
  28:     .eq('user_id', userId)
  29:     .eq('is_active', true)
  30: 
  31:   if (assignmentsError || !assignments || assignments.length === 0) {
  32:     return { isAdmin: false, permissions: [] }
  33:   }
  34: 
  35:   const roleIds = assignments.map((a: any) => a.role_id)
  36: 
  37:   const { data: permissionRows, error: permError } = await supabase
  38:     .from('role_permissions')
  39:     .select('permission_key')
  40:     .in('role_id', roleIds)
  41: 
  42:   if (permError || !permissionRows) {
  43:     return { isAdmin: false, permissions: [] }
  44:   }
  45: 
  46:   const keys = new Set<string>()
  47:   for (const p of permissionRows) {
  48:     if (p.permission_key) keys.add(p.permission_key)
  49:   }
  50: 
  51:   return { isAdmin: false, permissions: [...keys] }
  52: }

######## src\lib\supabase\authHelpers.ts ########
   1: import type { SupabaseClient, User } from '@supabase/supabase-js'
   2: 
   3: /**
   4:  * Supabase refresh tokens are single-use. When two requests share the same
   5:  * expiring session cookie and both attempt to refresh at nearly the same
   6:  * moment, the second one always fails with this error -- it's an expected,
   7:  * transient race under Supabase's model, not evidence of a real auth
   8:  * problem. Treat it as "try again shortly," never as "user is logged out."
   9:  */
  10: export function isTransientRefreshTokenError(
  11:   error: { message?: string | null; code?: string | null } | null | undefined
  12: ): boolean {
  13:   if (!error) return false
  14:   return (
  15:     error.code === 'refresh_token_already_used' ||
  16:     error.code === 'refresh_token_not_found' ||
  17:     !!error.message?.includes('Already Used') ||
  18:     !!error.message?.includes('Refresh Token Not Found')
  19:   )
  20: }
  21: 
  22: export type AuthResult =
  23:   | { user: User; transient: false }
  24:   | { user: null; transient: true }
  25:   | { user: null; transient: false }
  26: 
  27: /**
  28:  * Drop-in replacement for `supabase.auth.getUser()` in API routes and
  29:  * server components. Distinguishes a genuine "not logged in" from a
  30:  * transient refresh-token race, so callers can respond appropriately
  31:  * instead of hard-failing on a benign timing collision.
  32:  */
  33: export async function getAuthenticatedUser(supabase: SupabaseClient): Promise<AuthResult> {
  34:   try {
  35:     const { data, error } = await supabase.auth.getUser()
  36: 
  37:     if (error) {
  38:       if (isTransientRefreshTokenError(error)) {
  39:         return { user: null, transient: true }
  40:       }
  41:       return { user: null, transient: false }
  42:     }
  43: 
  44:     if (!data.user) {
  45:       return { user: null, transient: false }
  46:     }
  47: 
  48:     return { user: data.user, transient: false }
  49:   } catch (err: any) {
  50:     if (isTransientRefreshTokenError(err)) {
  51:       return { user: null, transient: true }
  52:     }
  53:     throw err
  54:   }
  55: }

  10: export async function requestDowngrade(formData: FormData) {
  11:   const supabase = await createClient()
  12:   const { user } = await getAuthenticatedUser(supabase)
  13:   if (!user) redirect('/login')
  14: 
  15:   const targetPlan = formData.get('target_plan') as PlanKey
  16: 
  17:   const { data: profile } = await supabase
  18:     .from('users').select('organization_id, role').eq('id', user.id).single()
  19: 
  20:   const isAdmin = profile?.role === 'admin' || profile?.role === 'school_admin'
  21:   const orgId = profile?.organization_id
  22: 
  23:   if (orgId && !isAdmin) return { success: false, message: 'Only administrators can change the subscription plan' }
  24: 
  25:   let usage = { students: 0, teachers: 0, classes: 0 }
  26: 
  27:   if (orgId) {
  28:     const { count: students } = await supabase.from('learners').select('*', { count: 'exact', head: true }).eq('organization_id', orgId)
  29:     const { count: teachers } = await supabase.from('users').select('*', { count: 'exact', head: true }).eq('organization_id', orgId).eq('role', 'teacher')
  30:     const { count: classes } = await supabase.from('groups').select('*', { count: 'exact', head: true }).eq('organization_id', orgId)
  31:     usage = { students: students ?? 0, teachers: teachers ?? 0, classes: classes ?? 0 }
  32:   } else {
  33:     const { count: students } = await supabase.from('learners').select('*', { count: 'exact', head: true }).eq('instructor_id', user.id)
  34:     const { count: classes } = await supabase.from('groups').select('*', { count: 'exact', head: true }).eq('instructor_id', user.id)
  35:     usage = { students: students ?? 0, teachers: 1, classes: classes ?? 0 }
  36:   }
  37: 
  38:   const { eligible, checks } = checkDowngradeEligibility(targetPlan, usage)
  39: 
  40:   if (!eligible) {
  41:     return {
  42:       success: false,
  43:       message: 'Your current usage exceeds the limits of that plan.',
  44:       checks: checks.filter(c => c.overLimit),
  45:     }
  46:   }
  47: 
  48:   const table = orgId ? 'organizations' : 'users'
  49:   const idColumn = orgId ? orgId : user.id
  50: 
  51:   const { error } = await supabase
  52:     .from(table)
  53:     .update({ subscription_plan: targetPlan })
  54:     .eq('id', idColumn)
  55: 
  56:   if (error) {
  57:     console.error('Downgrade error:', error)
  58:     return { success: false, message: 'Failed to change plan' }
  59:   }
  60: 
  61:   revalidatePath('/settings')
  62:   return { success: true }
  63: }

  35:       .update({ subscription_status: 'grace_period', grace_period_ends_at: graceEnd.toISOString() })

######## src\lib\subscription\getPlanFeatures.ts ########
   1: // src/lib/subscription/getPlanFeatures.ts
   2: import { SupabaseClient } from '@supabase/supabase-js'
   3: 
   4: /**
   5:  * Real source of truth for plan-based feature access, backed by the
   6:  * plan_features table, MERGED with any active feature_overrides for the
   7:  * organization (e.g. Founding 500 promotional grants, or any other
   8:  * manually-granted feature). Without this merge, feature_overrides rows
   9:  * are written correctly by enroll_founding_500() but silently ignored by
  10:  * every UI surface that calls this function -- confirmed root cause of
  11:  * Founding 500 schools seeing locked/upgrade-prompt feature cards despite
  12:  * having valid override grants in the database.
  13:  */
  14: export async function getPlanFeatures(
  15:   supabase: SupabaseClient,
  16:   planKey: string | null | undefined,
  17:   organizationId?: string | null
  18: ): Promise<string[]> {
  19:   const planFeaturesPromise = planKey
  20:     ? supabase.from('plan_features').select('feature_key').eq('plan_tier', planKey).eq('is_enabled', true)
  21:     : Promise.resolve({ data: [], error: null })
  22: 
  23:   const overridesPromise = organizationId
  24:     ? supabase
  25:         .from('feature_overrides')
  26:         .select('feature_key, enabled, expires_at')
  27:         .eq('organization_id', organizationId)
  28:     : Promise.resolve({ data: [], error: null })
  29: 
  30:   const [{ data: planFeatures, error: planError }, { data: overrides, error: overrideError }] =
  31:     await Promise.all([planFeaturesPromise, overridesPromise])
  32: 
  33:   const baseKeys = new Set<string>(planError || !planFeatures ? [] : planFeatures.map((r) => r.feature_key))
  34: 
  35:   if (!overrideError && overrides) {
  36:     const now = Date.now()
  37:     for (const o of overrides) {
  38:       const stillActive = !o.expires_at || new Date(o.expires_at).getTime() > now
  39:       if (o.enabled && stillActive) {
  40:         baseKeys.add(o.feature_key)
  41:       } else {
  42:         // An override can also explicitly REVOKE a feature the plan would
  43:         // otherwise grant (enabled: false) -- honor that by removing it,
  44:         // even though no current caller in this codebase does that yet.
  45:         baseKeys.delete(o.feature_key)
  46:       }
  47:     }
  48:   }
  49: 
  50:   return Array.from(baseKeys)
  51: }
